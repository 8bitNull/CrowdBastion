from pathlib import Path
import hashlib
import json
import re
import subprocess
from urllib.parse import unquote

folder = Path('docs/audit/2026-11-01-telegram-alignment')
results = []
shells = [Path('crowdbastion.sh'), *sorted(Path('lib').glob('*.sh'))]
for source in shells:
    subprocess.run(['bash', '-n', str(source)], check=True, capture_output=True)
results.append(f'Bash syntax: {len(shells)} files passed')
python_files = sorted(Path('lib').glob('*.py')) + sorted(Path('tests').glob('*.py'))
for source in python_files:
    compile(source.read_bytes(), str(source), 'exec')
results.append(f'Python compilation without bytecode writes: {len(python_files)} files passed')
related = ['crowdbastion.sh', 'lib/user_experience.sh', 'lib/node_guidance.sh', 'lib/health_summary.sh',
    'lib/private_transport.sh', 'lib/relay.sh', 'lib/v2bx.sh', 'lib/firewall.sh', 'lib/custom_bot.sh']
result = subprocess.run(['shellcheck', '-x', '-S', 'warning', *related], capture_output=True, text=True)
(folder / 'shellcheck.txt').write_text(result.stdout + result.stderr + f'Exit: {result.returncode}\n')
result.check_returncode()
results.append(f'ShellCheck warning level: all {len(related)} related Bash files passed')
result = subprocess.run(['bash', 'crowdbastion.sh', '--help'], capture_output=True, text=True, check=True)
(folder / 'help.txt').write_text(result.stdout)
for option in ('--status-json', '--custom-bot', '--runtime-status', '--sync-runtime'):
    assert option in result.stdout, option
results.append('Actual CLI help: exit 0, all four status/bot/runtime options present')
documents = [Path(name) for name in ('readme.md', 'docs/cli-reference.md', 'docs/usage-guide.md',
    'docs/2026-10-13-custom-telegram-bot.md', 'docs/audit/README.md',
    'docs/audit/2026-11-01-telegram-alignment.md',
    'docs/superpowers/specs/2026-11-01-telegram-alignment-design.md',
    'docs/superpowers/plans/2026-11-01-telegram-alignment.md')]
count = 0
for document in documents:
    for match in re.finditer(r'(?<!!)\[[^\]]*\]\(([^)]+)\)', document.read_text()):
        link = match[1].strip('<>')
        if re.match(r'^[a-zA-Z]+:', link):
            continue
        filename = unquote(link.split('#', 1)[0])
        target = document.parent / filename if filename else document
        assert target.exists(), (document, link)
        if '代码版本检查与更新' in link:
            fragment = link.split('#', 1)[1]
            headings = [re.sub(r'[^\w\s-]', '', value.lower()).replace(' ', '-')
                for value in re.findall(r'^#+\s+(.+)$', target.read_text(), re.M)]
            assert fragment in headings, (document, fragment)
        count += 1
results.append(f'Local Markdown links: {len(documents)} documents, {count} targets passed; new bot section anchors passed')
subprocess.run(['git', 'diff', '--check'], check=True, capture_output=True)
results.append('git diff --check: passed')
(folder / 'checks.txt').write_text('\n'.join(results)+'\n')
sources = sorted(set(shells + python_files + list(Path('lib').glob('*.lua'))))
manifest = {str(path): hashlib.sha256(path.read_bytes()).hexdigest() for path in sources}
previous = json.loads((folder / 'source-sha256.json').read_text()) if (folder / 'source-sha256.json').exists() else None
(folder / 'source-sha256.json').write_text(json.dumps(manifest, indent=2)+'\n')
print('\n'.join(results))
print(f'Source manifest: {len(sources)} files; unchanged since prior check: {manifest == previous}')
