from pathlib import Path
import json
import os
import subprocess
import time
root = Path('/projects/script/CrowdBastion')
folder = root/'docs/audit/2026-11-01-vps179-review'
command = ['python3', '-m', 'unittest', 'discover', '-s', 'tests']
start = time.time()
with (folder/'final-full.txt').open('w') as stream:
    result = subprocess.run(command, cwd=root, stdout=stream, stderr=subprocess.STDOUT,
        env=dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONPATH='tests:lib'))
(folder/'final-full-exit.txt').write_text(str(result.returncode)+'\n')
(folder/'final-full-run.json').write_text(json.dumps({'command': command, 'started_at': start,
    'completed_at': time.time(), 'exit': result.returncode}, indent=2)+'\n')
