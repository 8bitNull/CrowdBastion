import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import time


def command(argv, timeout=20):
    try:
        result = subprocess.run(argv, capture_output=True, text=True, timeout=timeout, env=dict(os.environ, LC_ALL='C', PYTHONDONTWRITEBYTECODE='1'))
        return {'exit': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr}
    except (OSError, subprocess.TimeoutExpired) as error:
        return {'error': type(error).__name__}


def canonical(text):
    tables = {}
    current = None
    for raw in text.splitlines():
        line = re.sub(r'\[\d+:\d+\]', '[0:0]', raw)
        if not line or line.startswith('#'):
            continue
        if line.startswith('*'):
            current = line
            tables[current] = {'chains': {}, 'rules': []}
        elif current and not re.search('crowdsec|docker', line, re.I):
            if line.startswith(':'):
                tables[current]['chains'][line.split()[0]] = line
            elif line != 'COMMIT':
                tables[current]['rules'].append(line)
    data = json.dumps(tables, sort_keys=True).encode()
    return {'sha256': hashlib.sha256(data).hexdigest(), 'tables': sorted(tables), 'static_rule_count': sum(len(table['rules']) for table in tables.values())}
