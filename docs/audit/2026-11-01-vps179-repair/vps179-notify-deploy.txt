from pathlib import Path
import hashlib,json,sqlite3,subprocess,sys,time
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
sys.path.insert(0,str(release/'lib'))
from crowdbastion_ux import redact
from crowdsec_notify import load_config
from crowdsec_notify_client import request
config=Path('/etc/crowdsec/crowdbastion-notify.json')
original=config.read_bytes()
parsed_before=load_config(config)
local_before=request(parsed_before,'status')
assert local_before['pending']==0 and local_before['retrying']==0 and local_before['open_critical']==0, 'queue requires separate review'
protected=[Path(p) for p in json.loads((backup/'protected-configs.before.json').read_text())]
hashes_before={str(path):hashlib.sha256(path.read_bytes()).hexdigest() for path in protected}

def snapshot():
    result={}
    for path in Path(parsed_before['state_dir']).glob('*.sqlite3'):
        with sqlite3.connect(path.as_uri()+'?mode=ro',uri=True) as db:
            assert db.execute('PRAGMA quick_check').fetchone()==('ok',)
            result[path.name]={'inode':path.stat().st_ino,'incidents':{row[0] for row in db.execute('SELECT id FROM incidents')},
                               'events':{row[0] for row in db.execute('SELECT fingerprint FROM events')}}
    return result
before=snapshot()
started=time.time()
command=subprocess.run(['/bin/bash',str(release/'crowdbastion.sh'),'--telegram-smart'],capture_output=True,text=True,timeout=40)
(backup/'notify-update-output.txt').write_text(command.stdout+'\n'+command.stderr)
assert command.returncode==0, 'notify installer failed; inspect private output'
assert load_config(config)==parsed_before,'notify settings changed'
assert config.read_bytes()==original,'notify config raw bytes changed'
after=snapshot()
assert before.keys()==after.keys()
for name,item in before.items():
    assert item['inode']==after[name]['inode']
    assert item['incidents']<=after[name]['incidents'] and item['events']<=after[name]['events']
helpers=['crowdsec_notify.py','crowdsec_notify_client.py','crowdsec_bot_api.py','crowdsec_bot_health.py','crowdbastion_health.py','crowdbastion_node.py','crowdsec_bot_runtime.py','crowdbastion_ux.py']
assert all((Path('/usr/local/lib/crowdbastion-notify')/name).read_bytes()==(release/'lib'/name).read_bytes() for name in helpers)
# Isolated interpreter: the installation alone must provide all imports.
runner='import runpy,sys;from pathlib import Path;sys.path.insert(0,str(Path(sys.argv[1]).parent));sys.argv=sys.argv[1:];runpy.run_path(sys.argv[0],run_name="__main__")'
check=subprocess.run(['python3','-I','-B','-c',runner,'/usr/local/lib/crowdbastion-notify/crowdsec_notify.py','check',str(config)],capture_output=True,text=True,timeout=10)
assert check.returncode==0
hashes_after={str(path):hashlib.sha256(path.read_bytes()).hexdigest() for path in protected}
assert hashes_before==hashes_after,'protected config drift'
service=subprocess.run(['systemctl','show','crowdbastion-notify.service','--property=ActiveState,UnitFileState,MainPID,NRestarts'],capture_output=True,text=True,check=True).stdout
values=dict(line.split('=',1) for line in service.splitlines() if '=' in line)
assert values['ActiveState']=='active' and values['UnitFileState']=='enabled'
report={'started_at':started,'completed_at':time.time(),'helper_files':len(helpers),'isolated_check_exit':check.returncode,'config_bytes_preserved':True,
        'protected_configs_preserved':True,'queue_files':[{'name':name,'inode_preserved':True,'incidents_retained':len(item['incidents']),'events_retained':len(item['events'])} for name,item in before.items()],
        'before_status':local_before,'after_status':request(parsed_before,'status'),'service':values,'update_output':redact(command.stdout+'\n'+command.stderr)}
(backup/'notify-deployment.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(report,ensure_ascii=False,indent=2))
