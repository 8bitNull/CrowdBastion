from pathlib import Path
import json,subprocess,sys,time
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
sys.path.insert(0,str(release/'lib'))
from crowdsec_firewall_evidence import inspect
from crowdsec_firewall_runtime import cron_risks,restore_services,chain_coverage
from crowdsec_existing_deployment import cron_list
from crowdsec_notify import load_config
from crowdsec_notify_client import request
config=Path('/etc/crowdsec/bouncers/crowdsec-firewall-bouncer.yaml')
report={'checked_at':time.time(),'decision_membership':inspect(config,True),'cron_risks':cron_risks(),'restore_services':restore_services(),'remaining_restore_crons':cron_list(),'chain_coverage':chain_coverage(config,True)}
assert report['decision_membership']['state']=='ok' and report['decision_membership']['missing']==0
assert report['cron_risks']==[] and report['restore_services']==[] and report['remaining_restore_crons']==[] and report['chain_coverage']=='ok'
report['notify_status']=request(load_config('/etc/crowdsec/crowdbastion-notify.json'),'status')
assert report['notify_status']['state']=='ready' and report['notify_status']['health_age']<60
report['uptime']=float(Path('/proc/uptime').read_text().split()[0])
report['services']={}
for name in ('crowdsec','crowdsec-firewall-bouncer','docker','crowdsec-1key-bot','crowdbastion-notify'):
    result=subprocess.run(['systemctl','show',name,'--property=ActiveState,MainPID,NRestarts'],capture_output=True,text=True,check=True,timeout=5)
    report['services'][name]=dict(line.split('=',1) for line in result.stdout.splitlines() if '=' in line)
    assert report['services'][name]['ActiveState']=='active' and report['services'][name]['NRestarts']=='0'
print(json.dumps(report,indent=2))
