from pathlib import Path
import hashlib,json,os,sqlite3,tempfile,time
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
path=backup/'backup-manifest.json'
original=path.read_bytes()
manifest=json.loads(original)
missing=[]
for name,digest in manifest.items():
    file=backup/name
    if not file.exists():
        assert file.parent==backup/'notify-consistent' and file.name.endswith(('.sqlite3-wal','.sqlite3-shm'))
        missing.append(name)
    else:assert hashlib.sha256(file.read_bytes()).hexdigest()==digest
assert len(missing)==2
for dbfile in (backup/'notify-consistent').glob('*.sqlite3'):
    connection=sqlite3.connect(dbfile.as_uri()+'?mode=ro&immutable=1',uri=True)
    try:assert connection.execute('PRAGMA quick_check').fetchone()==('ok',)
    finally:connection.close()
    assert hashlib.sha256(dbfile.read_bytes()).hexdigest()==manifest[str(dbfile.relative_to(backup))]
initial=backup/'backup-manifest.initial.json'
with initial.open('xb') as stream:stream.write(original)
initial.chmod(0o600)
sealed={name:digest for name,digest in manifest.items() if name not in missing}
fd,temporary=tempfile.mkstemp(prefix='.sealed-manifest-',dir=backup)
with os.fdopen(fd,'w') as stream:
    json.dump(sealed,stream,indent=2);stream.write('\n');stream.flush();os.fsync(stream.fileno())
os.replace(temporary,path)
report={'sealed_at':time.time(),'removed_transient_entries':missing,'initial_manifest_sha256':hashlib.sha256(original).hexdigest(),
        'sealed_manifest_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'stable_files':len(sealed),'all_stable_hashes_unchanged':True,'self_contained_database_quick_check':'ok'}
(backup/'backup-seal-correction.json').write_text(json.dumps(report,indent=2)+'\n')
(backup/'backup-seal-correction.json').chmod(0o600)
print(json.dumps(report,indent=2))
