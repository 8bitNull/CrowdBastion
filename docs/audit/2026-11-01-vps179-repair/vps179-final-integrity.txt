from pathlib import Path
import hashlib,json,os,re,subprocess,sys,time
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
sys.path.insert(0,str(release/'lib'))
from crowdsec_bot_runtime import maintenance_guard
from crowdbastion_ux import code_fingerprint
from crowdsec_notify import load_config
from crowdsec_notify_client import request
report={'checked_at':time.time()}
def run(argv):return subprocess.run(argv,capture_output=True,text=True,check=True,timeout=15).stdout

def normalized(text):
    return '\n'.join(re.sub(r'\[\d+:\d+\]','[0:0]',line) for line in text.splitlines() if line and not line.startswith('#'))
with maintenance_guard('/run/lock/crowdsec-1key.lock','/var/lib/crowdbastion/firewall-migration'):
    before={binary:normalized(run([binary])) for binary in ('iptables-save','ip6tables-save')}
    run(['systemctl','restart','crowdbastion-static-policy.service'])
    after={binary:normalized(run([binary])) for binary in before}
    assert before==after,'policy apply is not idempotent'
    report['policy_idempotent']=True
    report['static_rules']={}
    for binary in ('iptables','ip6tables'):
        rows=[line for line in run([binary,'-S','INPUT']).splitlines() if line.startswith('-A') and not re.search('CROWDSEC|DOCKER',line,re.I)]
        assert len(rows)==2
        assert all('! -i lo' in row and '-j DROP' in row for row in rows)
        assert sum('--dport 81 ' in row for row in rows)==1 and sum('--dport 3000 ' in row for row in rows)==1
        report['static_rules'][binary]=rows
    report['fingerprints']={str(path):code_fingerprint(path) for path in (release/'crowdbastion.sh',Path('/root/crowdsec-1key/crowdbastion.sh'),Path('/usr/local/lib/crowdsec-1key-bot/runtime/current/manager.sh'))}
    assert len(set(report['fingerprints'].values()))==1
    report['protected_config_hashes']={path:hashlib.sha256(Path(path).read_bytes()).hexdigest() for path in json.loads((backup/'protected-configs.before.json').read_text())}
    assert report['protected_config_hashes']==json.loads((backup/'protected-configs.before.json').read_text())
    for root in (release,Path('/root/crowdsec-1key')):
        assert root.stat().st_uid==0 and not root.stat().st_mode&0o022
        for path in root.rglob('*'):
            if path.is_symlink():
                assert path.name=='install-crowdsec-1panel.sh' and os.readlink(path)=='crowdbastion.sh'
            else:assert path.stat().st_uid==0 and not path.stat().st_mode&0o022
    report['source_ownership_and_modes']='ok'
    report['notify_status']=request(load_config('/etc/crowdsec/crowdbastion-notify.json'),'status')
    assert report['notify_status']['state']=='ready'
    assert report['notify_status']['pending']==0 and report['notify_status']['retrying']==0 and report['notify_status']['open_critical']==0
    report['services']={}
    baseline=json.loads((backup/'services.before.json').read_text())
    for name in ('crowdsec','crowdsec-firewall-bouncer','docker','crowdsec-1key-bot','crowdbastion-notify','crowdbastion-static-policy'):
        report['services'][name]=dict(line.split('=',1) for line in run(['systemctl','show',name,'--property=ActiveState,UnitFileState,MainPID,NRestarts,ExecMainStartTimestamp']).splitlines() if '=' in line)
        assert report['services'][name]['ActiveState']=='active' and report['services'][name]['NRestarts']=='0'
    report['planned_restarts']=[name for name in baseline if name in report['services'] and baseline[name]['MainPID']!=report['services'][name]['MainPID']]
    assert report['planned_restarts']==['crowdsec','crowdsec-1key-bot','crowdbastion-notify'],report['planned_restarts']
    report['bouncer_and_docker_pids_preserved']=True
    report['static_unit_order']=run(['systemctl','show','crowdbastion-static-policy.service','--property=Before,After']).strip()
    report['restore_service_state']=run(['systemctl','show','netfilter-persistent.service','--property=ActiveState,UnitFileState']).strip()
    for name,digest in json.loads((backup/'backup-manifest.json').read_text()).items():assert hashlib.sha256((backup/name).read_bytes()).hexdigest()==digest
    report['sealed_backup_manifest_valid']=True
    report['initial_backup_manifest_sha256']=hashlib.sha256((backup/'backup-manifest.initial.json').read_bytes()).hexdigest()
    report['completed_at']=time.time()
print(json.dumps(report,ensure_ascii=False,indent=2))
