from pathlib import Path
import hashlib, json, os, sqlite3, stat, subprocess, tarfile, time
os.umask(0o077)
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
for parent in (backup.parent, *backup.parent.parents):
    if parent.exists():
        info=parent.lstat()
        assert stat.S_ISDIR(info.st_mode) and info.st_uid==0 and not info.st_mode&0o022
backup.mkdir(parents=True,mode=0o700)
assert backup.stat().st_uid==0 and stat.S_IMODE(backup.stat().st_mode)==0o700

def run(argv):
    result=subprocess.run(argv,capture_output=True,timeout=25)
    assert result.returncode==0, ('command failed',argv[0],result.returncode)
    return result.stdout
paths=['/root/crowdsec-1key','/etc/crowdsec','/etc/systemd/system','/etc/cron.d',
       '/etc/iptables','/usr/lib/systemd/system/netfilter-persistent.service',
       '/usr/local/lib/crowdsec-1key-bot','/usr/local/lib/crowdbastion-notify',
       '/var/lib/crowdsec-1key-bot','/var/lib/crowdbastion']
with tarfile.open(backup/'configuration-code-and-bot-state.tar.gz','w:gz') as package:
    for name in paths:
        if Path(name).exists(): package.add(name,arcname=name.lstrip('/'))
(backup/'root.crontab.before').write_bytes(run(['crontab','-u','root','-l']))
for binary,name in [('iptables-save','live.v4'),('ip6tables-save','live.v6')]:
    (backup/name).write_bytes(run([binary]))
services={}
for unit in ('crowdsec','crowdsec-firewall-bouncer','docker','crowdsec-1key-bot','crowdbastion-notify','netfilter-persistent'):
    text=run(['systemctl','show',unit,'--property=Id,Names,MainPID,NRestarts,ActiveState,UnitFileState,FragmentPath']).decode()
    services[unit]=dict(line.split('=',1) for line in text.splitlines() if '=' in line)
(backup/'services.before.json').write_text(json.dumps(services,indent=2)+'\n')
databases=[]
state=Path('/var/lib/crowdbastion-notify')
(backup/'notify-consistent').mkdir(mode=0o700)
for path in state.glob('*.sqlite3'):
    assert path.is_file() and not path.is_symlink()
    target=backup/'notify-consistent'/path.name
    with sqlite3.connect(path.as_uri()+'?mode=ro',uri=True,timeout=5) as source:
        with sqlite3.connect(target) as destination:
            source.backup(destination)
            assert destination.execute('PRAGMA quick_check').fetchone()==('ok',)
    databases.append({'name':path.name,'quick_check':'ok','bytes':target.stat().st_size})
protected={name:hashlib.sha256(Path(name).read_bytes()).hexdigest() for name in (
    '/etc/crowdsec/config.yaml','/etc/crowdsec/config.yaml.local','/etc/crowdsec/crowdsec-1key-bot.json',
    '/etc/crowdsec/crowdbastion-notify.json','/etc/crowdsec/bouncers/crowdsec-firewall-bouncer.yaml',
    '/etc/crowdsec/bouncers/crowdsec-firewall-bouncer.yaml.local','/var/lib/crowdbastion/node-roles.json') if Path(name).is_file()}
(backup/'protected-configs.before.json').write_text(json.dumps(protected,indent=2)+'\n')
manifest={str(path.relative_to(backup)):hashlib.sha256(path.read_bytes()).hexdigest() for path in backup.rglob('*') if path.is_file()}
(backup/'backup-manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(json.dumps({'backup':str(backup),'archive_bytes':(backup/'configuration-code-and-bot-state.tar.gz').stat().st_size,
                  'consistent_databases':databases,'protected_config_hashes':protected,'services':services,
                  'backup_files':len(manifest),'completed_at':time.time()},indent=2))
