from pathlib import Path
import base64,hashlib,importlib.util,io,json,subprocess,tarfile
root=Path.cwd()
folder=root/'docs/audit/2026-11-01-vps179-repair'
# This utility archives the already reviewed evidence; it never reads SSH key contents.
spec=importlib.util.spec_from_file_location('repair_client','/tmp/vps179-repair-client.py')
client=importlib.util.module_from_spec(spec);spec.loader.exec_module(client)
buffer=io.BytesIO()
with tarfile.open(fileobj=buffer,mode='w:gz') as archive:
    for path in [root/'docs/audit/2026-11-01-vps179-repair.md',root/'docs/audit/README.md',*sorted(folder.iterdir())]:
        if path.is_file():archive.add(path,arcname=str(path.relative_to(root)),recursive=False)
payload=buffer.getvalue()
remote=r'''
from pathlib import Path
import base64,hashlib,io,json,os,stat,tarfile,time
import yaml
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
payload=base64.b64decode(__PAYLOAD__)
assert hashlib.sha256(payload).hexdigest()==__DIGEST__
secrets=set()
def walk(value):
    if isinstance(value,dict):
        for key,item in value.items():
            if str(key).lower() in ('bot_token','api_key','password','token','secret','api_secret') and isinstance(item,str) and len(item)>=12:secrets.add(item.encode())
            walk(item)
    elif isinstance(value,list):
        for item in value:walk(item)
for name in ('crowdsec-1key-bot.json','crowdbastion-notify.json','bouncers/crowdsec-firewall-bouncer.yaml','bouncers/crowdsec-firewall-bouncer.yaml.local','local_api_credentials.yaml','online_api_credentials.yaml'):
    path=Path('/etc/crowdsec')/name
    if path.is_file():walk(yaml.safe_load(path.read_text()))
leaks=[]
with tarfile.open(fileobj=io.BytesIO(payload),mode='r:gz') as archive:
    for member in archive:
        assert member.isfile() and not member.name.startswith('/') and '..' not in Path(member.name).parts
        data=archive.extractfile(member).read()
        if any(secret in data for secret in secrets):leaks.append(member.name)
assert not leaks,'credential value appears in audit artifact'
path=backup/'audit-review.tar.gz'
with path.open('xb') as stream:stream.write(payload)
path.chmod(0o600)
receipt={'completed_at':time.time(),'bundle_path':str(path),'bundle_sha256':hashlib.sha256(payload).hexdigest(),
         'credential_values_checked':len(secrets),'credential_value_leaks':leaks}
receipt_path=backup/'audit-delivery.json'
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n');receipt_path.chmod(0o600)
files={}
for path in sorted(backup.rglob('*')):
    relative=str(path.relative_to(backup))
    if relative=='final-manifest.json':continue
    if path.is_symlink():files[relative]={'type':'symlink','target':os.readlink(path)}
    elif path.is_file():files[relative]={'type':'regular','sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'bytes':path.stat().st_size}
manifest_path=backup/'final-manifest.json'
manifest_path.write_text(json.dumps({'sealed_at':time.time(),'files':files},indent=2)+'\n');manifest_path.chmod(0o600)
for name,item in files.items():
    if item['type']=='regular':assert hashlib.sha256((backup/name).read_bytes()).hexdigest()==item['sha256']
print(json.dumps({**receipt,'backup_manifest_path':str(manifest_path),'backup_manifest_sha256':hashlib.sha256(manifest_path.read_bytes()).hexdigest(),'backup_manifest_files':len(files)},indent=2))
'''
remote=remote.replace('__PAYLOAD__',repr(base64.b64encode(payload).decode())).replace('__DIGEST__',repr(hashlib.sha256(payload).hexdigest()))
result=subprocess.run(client.SSH+['env PYTHONDONTWRITEBYTECODE=1 python3 -B -'],input=remote.encode(),capture_output=True,timeout=55)
assert result.returncode==0,result.stderr.decode(errors='replace')[-1500:]
receipt=folder/'remote-seal.json';receipt.write_bytes(result.stdout);receipt.chmod(0o600)
print(result.stdout.decode())
