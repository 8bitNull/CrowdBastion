from pathlib import Path
import hashlib,json,sqlite3
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
manifest=json.loads((backup/'backup-manifest.json').read_text())
report={'missing':[],'mismatched':[],'databases':[]}
for name,digest in manifest.items():
    path=backup/name
    if not path.exists():report['missing'].append(name)
    elif hashlib.sha256(path.read_bytes()).hexdigest()!=digest:report['mismatched'].append(name)
for path in (backup/'notify-consistent').glob('*.sqlite3'):
    connection=sqlite3.connect(path.as_uri()+'?mode=ro&immutable=1',uri=True)
    try:
        report['databases'].append({'name':path.name,'quick_check':connection.execute('PRAGMA quick_check').fetchone()[0],
                                   'incidents':connection.execute('SELECT COUNT(*) FROM incidents').fetchone()[0],'events':connection.execute('SELECT COUNT(*) FROM events').fetchone()[0]})
    finally:connection.close()
print(json.dumps(report,indent=2))
