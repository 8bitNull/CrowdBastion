from pathlib import Path
import hashlib,json,os,stat,subprocess,sys,time
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
sys.path.insert(0,str(release/'lib'))
from crowdsec_bot_runtime import maintenance_guard
import crowdsec_existing_deployment as existing
import crowdsec_firewall_runtime as runtime
from crowdbastion_ux import validate_private_destination
os.umask(0o077)
unit=Path('/etc/systemd/system/crowdbastion-static-policy.service')
code=Path('/usr/local/lib/crowdbastion-static-policy/apply.py')
validate_private_destination(unit)
validate_private_destination(code,missing_parents=True)
assert not unit.exists() and not unit.is_symlink() and not code.parent.exists()
rules=[(binary,['INPUT','!','-i','lo','-p','tcp','--dport',port,'-j','DROP']) for binary in ('/usr/sbin/iptables','/usr/sbin/ip6tables') for port in ('81','3000')]

def run(argv):
    result=subprocess.run(argv,capture_output=True,text=True,timeout=20)
    if result.returncode: raise RuntimeError('command failed: '+argv[0])
    return result.stdout.strip()

def present(binary,rule):
    result=subprocess.run([binary,'-w','5','-C',*rule],capture_output=True,timeout=8)
    assert result.returncode in (0,1),('rule check failed',binary)
    return result.returncode==0

script='''# CrowdBastion managed: only NPM management ports, no whole-table restoration.
import signal
import subprocess

def interrupted(signum, frame):
    raise SystemExit(128+signum)
signal.signal(signal.SIGTERM, interrupted)
signal.signal(signal.SIGINT, interrupted)
added=[]
try:
    for binary in ('/usr/sbin/iptables','/usr/sbin/ip6tables'):
        for port in ('81','3000'):
            rule=['INPUT','!','-i','lo','-p','tcp','--dport',port,'-j','DROP']
            result=subprocess.run([binary,'-w','5','-C',*rule],capture_output=True,timeout=8)
            if result.returncode==1:
                subprocess.run([binary,'-w','5','-A',*rule],check=True,timeout=8)
                added.append((binary,rule))
            elif result.returncode:
                raise SystemExit(result.returncode)
except BaseException:
    signal.signal(signal.SIGTERM, signal.SIG_IGN)
    signal.signal(signal.SIGINT, signal.SIG_IGN)
    for binary,rule in reversed(added):
        subprocess.run([binary,'-w','5','-D',*rule],timeout=8)
    raise
'''
service='''# CrowdBastion managed static policy; NPM management through loopback/SSH tunnel
[Unit]
Description=CrowdBastion narrowly scoped NPM management firewall policy
After=local-fs.target
Before=docker.service crowdsec-firewall-bouncer.service

[Service]
Type=oneshot
ExecStart=/usr/bin/python3 -B /usr/local/lib/crowdbastion-static-policy/apply.py
RemainAfterExit=yes
UMask=0077

[Install]
WantedBy=multi-user.target
'''
transitions=[]
report={'started_at':time.time(),'backup':str(backup)}
with maintenance_guard('/run/lock/crowdsec-1key.lock','/var/lib/crowdbastion/firewall-migration'):
    initial=[present(binary,rule) for binary,rule in rules]
    assert initial==[True,False,False,False],initial
    assert existing._cron_read()==(backup/'root.crontab.before').read_bytes()
    rows=existing.cron_list()
    assert [row['family'] for row in rows]==['IPv4','IPv6'],rows
    report['cron_export_indexes']=[row['index'] for row in rows]
    assert run(['systemctl','is-enabled','netfilter-persistent.service'])=='enabled'
    try:
        code.parent.mkdir(mode=0o700)
        code.write_text(script);code.chmod(0o600)
        unit.write_text(service);unit.chmod(0o644)
        run(['systemd-analyze','verify',str(unit)])
        run(['systemctl','daemon-reload'])
        run(['systemctl','enable','--now',unit.name])
        assert all(present(binary,rule) for binary,rule in rules)
        for row in rows:
            target=backup/('cron-disable-'+str(row['index']))
            target.mkdir(mode=0o700)
            transitions.append(target)
            existing.cron_disable(target,row['index'])
        run(['systemctl','disable','netfilter-persistent.service'])
        run(['systemctl','daemon-reload'])
        assert existing.cron_list()==[]
        assert runtime.cron_risks()==[]
        assert runtime.restore_services()==[]
        assert runtime.chain_coverage(Path('/etc/crowdsec/bouncers/crowdsec-firewall-bouncer.yaml'),True)=='ok'
        report['cron_after_sha256']=hashlib.sha256(existing._cron_read()).hexdigest()
        report['restore_services']=runtime.restore_services()
        report['chain_coverage']='ok'
        report['static_policy']={'ports':[81,3000],'families':['ipv4','ipv6'],'allowed_interface':'lo','existing_ipv4_81_preserved':True,'added_rules':3}
        report['service']=dict(line.split('=',1) for line in run(['systemctl','show',unit.name,'--property=ActiveState,SubState,UnitFileState']).splitlines() if '=' in line)
        report['old_restore_service']=dict(line.split('=',1) for line in run(['systemctl','show','netfilter-persistent.service','--property=ActiveState,UnitFileState']).splitlines() if '=' in line)
        report['completed_at']=time.time()
        (backup/'static-policy-script.txt').write_text(script)
        (backup/'static-policy-unit.txt').write_text(service)
        (backup/'startup-repair.json').write_text(json.dumps(report,indent=2)+'\n')
    except BaseException:
        recovery=[]
        for transition in reversed(transitions):
            try: existing.cron_restore(transition)
            except Exception: recovery.append('cron restore failed')
        for command in (['systemctl','enable','netfilter-persistent.service'],['systemctl','disable','--now',unit.name]):
            if subprocess.run(command,capture_output=True).returncode: recovery.append('unit recovery failed')
        for existed,(binary,rule) in reversed(list(zip(initial,rules))):
            if not existed and present(binary,rule):
                if subprocess.run([binary,'-w','5','-D',*rule],capture_output=True).returncode: recovery.append('rule recovery failed')
        unit.unlink(missing_ok=True);code.unlink(missing_ok=True)
        subprocess.run(['systemctl','daemon-reload'],capture_output=True)
        (backup/'startup-recovery-errors.json').write_text(json.dumps(recovery)+'\n')
        raise
print(json.dumps(report,ensure_ascii=False,indent=2))
