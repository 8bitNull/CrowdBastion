from pathlib import Path
import json,os,stat
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
assert release.is_dir() and not release.is_symlink()
changes=[]
for path in [release,*sorted(release.rglob('*'))]:
    info=path.lstat()
    if stat.S_ISLNK(info.st_mode):
        assert path==release/'install-crowdsec-1panel.sh' and os.readlink(path)=='crowdbastion.sh'
        os.chown(path,0,0,follow_symlinks=False)
        continue
    assert stat.S_ISDIR(info.st_mode) or stat.S_ISREG(info.st_mode)
    mode=0o700 if path==release else 0o755 if path.is_dir() or path.name in ('crowdbastion.sh','install-crowdsec-1panel.sh') else 0o644
    os.chown(path,0,0)
    path.chmod(mode)
    changes.append({'path':str(path.relative_to(release)), 'before_mode':oct(stat.S_IMODE(info.st_mode)), 'before_uid':info.st_uid,'mode':oct(mode),'uid':0})
print(json.dumps({'normalized':changes},indent=2))
