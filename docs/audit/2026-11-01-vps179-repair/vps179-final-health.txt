from pathlib import Path
import hashlib,json,os,subprocess,sys,time
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
installed=Path('/usr/local/lib/crowdsec-1key-bot')
sys.path.insert(0,str(installed))
import crowdsec_bot as bot
from crowdsec_bot_runtime import maintenance_guard
bot.load_settings('/etc/crowdsec/crowdsec-1key-bot.json')
report={'checked_at':time.time(),'bot_manager':bot.MANAGER_SCRIPT,'maintenance_lock':bot.MAINTENANCE_LOCK}
start=time.monotonic()
terminal=subprocess.run(['/usr/local/bin/crowdbastion','--status-json'],capture_output=True,text=True,timeout=25)
report['terminal_seconds']=round(time.monotonic()-start,3)
report['terminal_exit']=terminal.returncode
report['terminal_report']=json.loads(terminal.stdout)
start=time.monotonic()
report['bot_rendered_health']=bot.module_health_status(details=True)
report['bot_seconds']=round(time.monotonic()-start,3)
report['bot_report']=bot._HEALTH_COLLECTOR[1]._cached
assert report['terminal_exit']==2
for key in ('checks','summary','scope'):
    assert report['terminal_report'][key]==report['bot_report'][key],key
assert report['bot_seconds']<20
runtime=bot.sibling('crowdsec_bot_api').read_runtime(bot.STATE_DIR,bot.BOT_TOKEN)
# Retain only timestamps/status, never the persisted API endpoint or token.
report['poller']={key:value for key,value in runtime.items() if key in ('status','last_poll_at','last_poll_attempt_at','last_send_at','last_error_at','last_error_kind','checked_at')}
assert runtime['status']=='healthy'
assert runtime['last_poll_at']>json.loads((backup/'bot-deployment.json').read_text())['started_at']
instances=[]
for process in Path('/proc').glob('[0-9]*'):
    try:
        arguments=(process/'cmdline').read_bytes().split(b'\0')
        if str(installed/'crowdsec_bot.py').encode() in arguments:instances.append(int(process.name))
    except OSError:pass
report['bot_instance_pids']=instances
assert len(instances)==1
with maintenance_guard(bot.MAINTENANCE_LOCK,bot.MIGRATION_STATE):
    refusal=subprocess.run(['/bin/bash','-c','source "$1"; locked_operation printf SHOULD_NOT_RUN','--',str(release/'crowdbastion.sh')],capture_output=True,text=True,timeout=10)
    assert refusal.returncode==75 and 'SHOULD_NOT_RUN' not in refusal.stdout
    report['cli_lock_refusal_exit']=refusal.returncode
    runner='import sys;sys.path.insert(0,sys.argv[1]);from crowdsec_bot_runtime import maintenance_guard,MaintenanceRefused\ntry:\n with maintenance_guard(sys.argv[2],sys.argv[3]):raise SystemExit(9)\nexcept MaintenanceRefused:raise SystemExit(75)'
    refusal=subprocess.run(['python3','-I','-B','-c',runner,str(installed),bot.MAINTENANCE_LOCK,bot.MIGRATION_STATE],capture_output=True,text=True,timeout=10)
    assert refusal.returncode==75
    report['bot_guard_refusal_exit']=refusal.returncode
protected=json.loads((backup/'protected-configs.before.json').read_text())
report['protected_config_hashes']={path:hashlib.sha256(Path(path).read_bytes()).hexdigest() for path in protected}
assert report['protected_config_hashes']==protected
report['protected_config_bytes_preserved']=True
report['services']={}
for name in ('crowdsec','crowdsec-firewall-bouncer','crowdsec-1key-bot','crowdbastion-notify','docker','crowdbastion-static-policy'):
    values=subprocess.run(['systemctl','show',name,'--property=MainPID,ActiveState,UnitFileState,NRestarts,ExecMainStartTimestamp'],capture_output=True,text=True,check=True).stdout
    report['services'][name]=dict(line.split('=',1) for line in values.splitlines() if '=' in line)
    assert report['services'][name]['ActiveState']=='active' and report['services'][name]['NRestarts']=='0'
print(json.dumps(report,ensure_ascii=False,indent=2))
