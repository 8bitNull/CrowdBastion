import json,subprocess,urllib.request,urllib.error
class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self,*args):return None
opener=urllib.request.build_opener(urllib.request.ProxyHandler({}),NoRedirect())
report={}
for address in ('127.0.0.1','[::1]'):
    for port in (81,3000):
        key=f'{address}:{port}'
        try:
            with opener.open(f'http://{key}/',timeout=3) as response:report[key]={'status':response.status}
        except urllib.error.HTTPError as error:report[key]={'status':error.code}
        except Exception as error:report[key]={'error':type(error).__name__}
result=subprocess.run(['docker','exec','npm','nginx','-T'],capture_output=True,text=True,timeout=15)
report['npm_configuration']={'exit':result.returncode,'api_uses_loopback_3000':bool(__import__('re').search(r'proxy_pass\s+http://(?:127\.0\.0\.1|localhost):3000',result.stdout))}
report['ipv6_global_addresses']=[row['local'] for interface in json.loads(subprocess.run(['ip','-j','-6','addr'],capture_output=True,text=True,check=True).stdout) for row in interface.get('addr_info',[]) if row.get('scope')=='global']
print(json.dumps(report,indent=2))
