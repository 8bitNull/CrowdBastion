from pathlib import Path
import json,os,hashlib
report={}
for name in ('/usr/local/bin/crowdbastion','/usr/local/bin/crowdbastion.sh','/root/crowdbastion.sh','/root/install-crowdsec-1panel.sh'):
    path=Path(name)
    if path.exists() or path.is_symlink():
        info=path.lstat()
        report[name]={'mode':oct(info.st_mode&0o7777),'uid':info.st_uid,'link':os.readlink(path) if path.is_symlink() else None,'sha256':hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None}
    else: report[name]={'exists':False}
print(json.dumps(report,indent=2))
