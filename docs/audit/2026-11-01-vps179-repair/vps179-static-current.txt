import subprocess,json,shlex
result={}
for binary in ('iptables','ip6tables'):
    rows=subprocess.run([binary,'-w','5','-S'],capture_output=True,text=True,check=True).stdout.splitlines()
    result[binary]=[line for line in rows if not any(s in line.upper() for s in ('CROWDSEC','DOCKER'))]
probe=subprocess.run(['iptables','-w','5','-C','INPUT','-i','lo','-p','tcp','--dport','81','-j','DROP'],capture_output=True,text=True)
result['exact_rule_check']={'exit':probe.returncode,'stderr':probe.stderr}
print(json.dumps(result,indent=2))
