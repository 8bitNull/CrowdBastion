from pathlib import Path
import hashlib,json,os,shutil,subprocess,sys,time
release=Path('/opt/crowdbastion/releases/review-20261101-vps179-29689b60aa54')
backup=Path('/var/backups/crowdbastion/vps179-cycle-20261101T123337Z')
root=Path('/root/crowdsec-1key')
next_root=Path('/root/.crowdsec-1key.next-20261101T123337Z')
entry=Path('/usr/local/bin/crowdbastion')
sys.path.insert(0,str(release/'lib'))
from crowdsec_bot_runtime import maintenance_guard
from crowdbastion_ux import code_fingerprint,validate_private_destination
os.umask(0o077)
expected='29689b60aa5410dc1baa296594fecd519be4f1b3e8d715bd5d2851804a34e573'
assert code_fingerprint(release/'crowdbastion.sh')==expected
validate_private_destination(entry)
assert not entry.exists() and not entry.is_symlink()
assert root.is_dir() and not root.is_symlink() and root.stat().st_uid==0
assert root.stat().st_dev==backup.stat().st_dev
assert not next_root.exists() and not (backup/'source-directory.before').exists()
with maintenance_guard('/run/lock/crowdsec-1key.lock','/var/lib/crowdbastion/firewall-migration'):
    shutil.copytree(release,next_root,symlinks=True)
    assert code_fingerprint(next_root/'crowdbastion.sh')==expected
    moved=False
    try:
        root.rename(backup/'source-directory.before');moved=True
        next_root.rename(root)
        with entry.open('x') as stream:
            stream.write('#!/bin/sh\n# CrowdBastion managed entry; execute the complete release.\nexec /bin/bash "'+str(release/'crowdbastion.sh')+'" "$@"\n')
        entry.chmod(0o755)
        result=subprocess.run([str(entry),'--help'],capture_output=True,text=True,timeout=10)
        assert result.returncode==0 and '--status-json' in result.stdout
        assert code_fingerprint(root/'crowdbastion.sh')==expected
    except BaseException:
        entry.unlink(missing_ok=True)
        if moved:
            if root.exists():root.rename(backup/'source-directory.failed')
            (backup/'source-directory.before').rename(root)
        raise
report={'release':str(release),'working_copy':str(root),'entrypoint':str(entry),'fingerprint':expected,'help_exit':0,'completed_at':time.time()}
(backup/'manager-deployment.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
