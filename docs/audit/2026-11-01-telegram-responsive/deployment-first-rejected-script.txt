from pathlib import Path
import hashlib,json,os,shutil,signal,subprocess,sys,time
os.umask(0o077)
meta=json.loads(Path('/tmp/crowdbastion-responsive-deployment.json').read_text())
release,backup=Path(meta['release']),Path(meta['backup'])
sys.path.insert(0,str(release/'lib'))
from crowdsec_bot_runtime import maintenance_guard
from crowdbastion_ux import code_fingerprint,redact,validate_private_destination
assert code_fingerprint(release/'crowdbastion.sh')==meta['fingerprint']
config=Path('/etc/crowdsec/crowdsec-1key-bot.json')
original=config.read_bytes();settings=json.loads(original)
state=Path(settings.get('state_dir','/var/lib/crowdsec-1key-bot'))
root=Path('/root/crowdsec-1key');entry=Path('/usr/local/bin/crowdbastion')
next_root=Path('/root/.crowdsec-1key.tg-responsive.next')
validate_private_destination(entry)
assert entry.read_bytes()==(backup/'entry.before').read_bytes()
assert not next_root.exists() and not (backup/'source-directory.before').exists()
assert code_fingerprint(root/'crowdbastion.sh')=='29689b60aa5410dc1baa296594fecd519be4f1b3e8d715bd5d2851804a34e573'
offsets={p.name:int(p.read_text()) for p in state.glob('offset-*')}
audits={p.name:p.read_bytes() for p in state.glob('bot-actions-*.jsonl*')}
services={}
for unit in ['crowdsec','crowdsec-firewall-bouncer','crowdsec-1key-bot','crowdbastion-notify','docker']:
    out=subprocess.check_output(['systemctl','show',unit,'--property=ActiveState,UnitFileState,MainPID,NRestarts'],text=True)
    services[unit]=dict(line.split('=',1) for line in out.splitlines() if '=' in line)
assert services['crowdsec-1key-bot']['ActiveState']=='active' and services['crowdsec-1key-bot']['UnitFileState']=='enabled'
started=time.time()
with maintenance_guard(settings.get('maintenance_lock','/run/lock/crowdsec-1key.lock'),settings.get('migration_state','/var/lib/crowdbastion/firewall-migration')):
    shutil.copytree(release,next_root,symlinks=True)
    assert code_fingerprint(next_root/'crowdbastion.sh')==meta['fingerprint']
    root_moved=False
    try:
        root.rename(backup/'source-directory.before');root_moved=True
        next_root.rename(root)
        temporary=entry.with_name('.crowdbastion.tg-responsive.next')
        with temporary.open('x') as out:
            out.write('#!/bin/sh\n# CrowdBastion managed entry; execute the complete release.\nexec /bin/bash "'+str(release/'crowdbastion.sh')+'" "$@"\n')
        temporary.chmod(0o755);temporary.replace(entry)
        command=subprocess.run(['/bin/bash',str(release/'crowdbastion.sh'),'--custom-bot'],input='5\ny\n0\n',capture_output=True,text=True,timeout=40)
        (backup/'bot-update-output.private.txt').write_text(command.stdout+'\n'+command.stderr)
        assert command.returncode==0,'product updater returned failure'
        assert config.read_bytes()==original,'configuration bytes changed'
        status=subprocess.run(['/bin/bash','-c','source "$1"; custom_bot_code_status','--',str(release/'crowdbastion.sh')],capture_output=True,text=True,timeout=10)
        assert status.returncode==0,'installed code does not match release'
    except BaseException:
        if root_moved:
            if root.exists():root.rename(backup/'source-directory.failed')
            (backup/'source-directory.before').rename(root)
        previous=entry.with_name('.crowdbastion.tg-responsive.restore')
        previous.write_bytes((backup/'entry.before').read_bytes());previous.chmod(0o755);previous.replace(entry)
        raise
assert all(int((state/name).read_text())>=offset for name,offset in offsets.items())
assert all((state/name).read_bytes().startswith(value) for name,value in audits.items())
protected=json.loads((backup/'protected.before.json').read_text())
assert all(hashlib.sha256(Path(name).read_bytes()).hexdigest()==digest for name,digest in protected.items())
after={}
for unit in services:
    out=subprocess.check_output(['systemctl','show',unit,'--property=ActiveState,UnitFileState,MainPID,NRestarts'],text=True)
    after[unit]=dict(line.split('=',1) for line in out.splitlines() if '=' in line)
    assert after[unit]['ActiveState']==services[unit]['ActiveState'] and after[unit]['UnitFileState']==services[unit]['UnitFileState']
    if unit!='crowdsec-1key-bot':assert after[unit]['MainPID']==services[unit]['MainPID']
report={'started_at':started,'finished_at':time.time(),'release':str(release),'backup':str(backup),'fingerprint':meta['fingerprint'],
        'config_bytes_preserved':True,'protected_configs_preserved':len(protected),'offsets_checked':len(offsets),'audits_checked':len(audits),
        'services_before':services,'services_after':after,'code_status_exit':status.returncode,'code_status':status.stdout,
        'update_output':redact(command.stdout+'\n'+command.stderr)}
(backup/'deployment-result.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(report,ensure_ascii=False,indent=2))
